	int fromHeight;
	int scaleStartTime;
	int scaleTime;

	float fromY;

	int fromAlignScreen;

	int moveTime;
	int time;
	int duration;
	float value;
	int text;
	float sort;

	int fxBirthTime;
	int fxLetterTime;
	int fxDecayStartTime;
	int fxDecayDuration;
	int soundID;
	int flags;
};
